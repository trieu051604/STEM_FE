const int RED = 13;
const int YELLOW = 12;
const int GREEN = 11;

void setup() {
  pinMode(RED, OUTPUT);
  pinMode(YELLOW, OUTPUT);
  pinMode(GREEN, OUTPUT);
}

void loop() {
  // Đèn xanh sáng 5 giây
  digitalWrite(GREEN, HIGH);
  delay(5000);
  digitalWrite(GREEN, LOW);

  // Đèn vàng sáng 2 giây

  delay(2000);
  digitalWrite(YELLOW, LOW);

  // Đèn đỏ sáng 5 giây

  delay(5000);
  digitalWrite(RED, LOW);
}
